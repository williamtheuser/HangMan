using System;

namespace HangMan
{
    class Program
    {
        static void Main(string[] args)
        {
            while (true)
            {
                string[] WordBase = ReadSecretWord();
                string WordToGuess = WordBase[0];    // word to guess
                string WordPlaceholder = WordBase[1]; // placeholder for the word
                
                int wrongGuessesCount = 0;
                string wrongGuessesChars = "";
                int currentFrame = 0;

                DisplayWord(WordPlaceholder);

                while (true)
                {
                    bool guessSuccess = false;
                    char guessedChar = ReadOneChar(); // Get player guess
                    EvaluateTheSituation(ref WordBase, guessedChar, ref wrongGuessesCount, ref wrongGuessesChars, ref guessSuccess);

                    if (wrongGuessesCount >= 6)
                    {
                        HangTheMan(currentFrame, wrongGuessesCount, wrongGuessesChars);
                        Console.WriteLine("Game over! You've lost.");
                        break;
                    }
                    
                    if (!WordBase[1].Contains("_"))
                    {
                        Console.WriteLine("Congratulations! You've guessed the word!");
                        break;
                    }

                    HangTheMan(currentFrame, wrongGuessesCount, wrongGuessesChars);
                }

                QuitOrRestart();
            }
        }

        static void DisplayWord(string word)
        {
            Console.Clear();
            Console.WriteLine($"Word to guess: {word}");
        }

        static string input(string message)
        {
            Console.WriteLine(message);
            return Console.ReadLine();
        }

        static string[] ReadSecretWord()
        {
            string allowedChars = "abcdefghijklmnopqrstuvwxyz";
            while (true)
            {
                string secretWord = input("Enter the secret word to guess (at least 4 letters, no numbers):");
                if (secretWord.Length < 4 || secretWord.Any(c => !allowedChars.Contains(c)))
                {
                    Console.WriteLine("Invalid input! The word must be at least 4 characters long and contain only letters.");
                    continue;
                }

                string wordPlaceholder = new string('_', secretWord.Length);
                return new string[] { secretWord, wordPlaceholder };
            }
        }

        static char ReadOneChar()
        {
            char letterIn;
            while (true)
            {
                Console.Write("Enter a letter: ");
                letterIn = Console.ReadKey().KeyChar;
                Console.WriteLine();

                if (char.IsLetter(letterIn))
                {
                    break;
                }
                else
                {
                    Console.WriteLine("Invalid input! Please enter a letter.");
                }
            }

            return letterIn;
        }

        static void EvaluateTheSituation(ref string[] WordBase, char guessedChar, ref int wrongGuessesCount, ref string wrongGuessesChars, ref bool guessSuccess)
        {
            string wordToGuess = WordBase[0];
            string currentPlaceholder = WordBase[1];
            
            if (!wordToGuess.Contains(guessedChar.ToString()))
            {
                wrongGuessesCount++;
                wrongGuessesChars += guessedChar + ", ";
                guessSuccess = false;
            }
            else
            {
                guessSuccess = true;
                char[] updatedPlaceholder = currentPlaceholder.ToCharArray();

                for (int i = 0; i < wordToGuess.Length; i++)
                {
                    if (wordToGuess[i] == guessedChar)
                    {
                        updatedPlaceholder[i] = guessedChar;
                    }
                }

                WordBase[1] = new string(updatedPlaceholder);
            }

            DisplayWord(WordBase[1]);
        }

        static void HangTheMan(int currentFrame, int wrongGuessesCount, string wrongGuessesChars)
        {
            string[] Frames = {
                @"
                ----+
                |   |
                    |
                    |
                    |
                ==========
                ",
                @"
                ----+
                |   |
                0   |
                    |
                    |
                ==========
                ",
                @"
                ----+
                |   |
                0   |
                |   |
                    |
                ==========
                ",
                @"
                ----+
                |   |
                0   |
               /|   |
                    |
                ==========
                ",
                @"
                ----+
                |   |
                0   |
               /|\  |
                    |
                ==========
                ",
                @"
                ----+
                |   |
                0   |
               /|\  |
               /    |
                ==========
                ",
                @"
                ----+
                |   |
                0   |
               /|\  |
               / \  |
                ==========
                "
            };

            Console.Clear();
            Console.WriteLine(Frames[wrongGuessesCount]);

            Console.WriteLine($"Wrong guesses: {wrongGuessesChars}");
        }

        static void QuitOrRestart()
        {
            string choice = input("Would you like to quit or restart the game? (Q to quit, any other key to restart)").ToUpper();

            if (choice == "Q")
            {
                Environment.Exit(0); // Exit the game
            }
        }
    }
}
